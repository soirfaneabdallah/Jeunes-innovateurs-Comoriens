	\section{C'est quoi Anaconda ?}
	
	\begin{center}
		\fbox{\parbox{0.9\textwidth}{
				\begin{minipage}{0.2\textwidth}
				%	\includegraphics[width=0.9\textwidth]{frigo.jpg}

						\item Des mangues (des bibliothèques comme NumPy, Pandas)

					\end{itemize}
				\end{minipage}
		}}
	\end{center}
	
	Anaconda, c'est comme un frigo géant pour programmeurs.
	
	Imaginez que vous allez au marché de Volo-Volo. Vous avez besoin de :
	\begin{itemize}
		\item Des bananes (Python)
		\item Des mangues (des bibliothèques comme NumPy, Pandas)
		\item Du manioc (Jupyter Notebook)
		\item Et un soda (Spyder)
	\end{itemize}
	
	Vous pourriez tout acheter séparément. Mais Youssouf, lui, il est malin. Il va chez \textbf{Anaconda Market}, un supermarché géant qui vend TOUT en un seul pack.
	
	\begin{definition}
		\textbf{Anaconda} est une distribution Python qui inclut :
		\begin{itemize}
			\item Python (le langage)
			\item Plus de 700 bibliothèques pré-installées (NumPy, Pandas, Matplotlib, Scikit-learn, TensorFlow, PyTorch...)
			\item Des outils de développement (Jupyter, Spyder, VS Code)
			\item Un gestionnaire d'environnements (pour ne pas tout casser)
		\end{itemize}
	\end{definition}
	
	\textbf{En gros : Anaconda, c'est comme si vous aviez un assistant personnel qui range votre code, installe vos bibliothèques, et ne se moque pas de vous quand vous faites une erreur.}

	\subsection*{Pour Windows}
	
	\begin{installation}
		\begin{enumerate}
			\item Ouvrez votre navigateur préféré (Chrome, Firefox, ou m\^eme Edge si vous êtes un aventurier)

			\item Cliquez dessus. \textbf{ATTENTION :} choisissez la version \textbf{Windows} (pas Mac, pas Linux)
			\item Le téléchargement commence. C'est un fichier de \textasciitilde 600-800 Mo (ça va durer le temps de boire un café... ou deux)
		\end{enumerate}

	\section*{Étape 2 : Lancer l'installation}
	\addcontentsline{toc}{section}{Étape 2 : Lancer l'installation}
	
	\subsection*{Cas Windows}
	
	\begin{installation}
		\begin{enumerate}
			\item Double-cliquez sur le fichier \texttt{Anaconda3-\ldots-Windows-x86\_64.exe}
			\item Votre ordinateur vous dit : "Ce programme va faire des changements sur votre ordinateur"
			\item Vous cliquez sur \textbf{OUI}. C'est votre premier vrai choix de programmation. Félicitations.
			
			\item \textbf{L'écran d'installation apparaît.} C'est vert. C'est beau. C'est Anaconda.
			\item Cliquez sur \textbf{Next} (Suivant)
			\item \textit{Conseil du samedi soir :} Lisez pas tout. Faites confiance.
			
			\item \textbf{Acceptez la licence.} 
			\begin{itemize}
				\item C'est un texte long. Personne ne le lit. C'est comme les CGU de WhatsApp. Cliquez sur "I Agree".
			\end{itemize}
			
			\item \textbf{Choisissez "Just Me"} (sauf si vous êtes administrateur et que vous voulez partager avec toute la classe... mais non, vous êtes égoïste, prenez "Just Me")
			
			\item \textbf{Choisissez le dossier d'installation}
			\begin{itemize}
				\item Par défaut : \texttt{C:\textbackslash Users\textbackslash VotreNom\textbackslash anaconda3}
				\item Gardez-le comme ça. Ne changez pas. Votre futur vous remerciera.
				\item \textit{Si vous changez, vous allez le regretter dans 3 mois quand vous chercherez où se cache ce fichier.}
			\end{itemize}
			
			\item \textbf{LE MOMENT CRUCIAL}
			
			Vous arrivez à un écran avec deux options (c'est là que tout se joue) :
			
			\begin{enumerate}
				\item \textbf{Non} à "Add Anaconda3 to my PATH environment variable"
				\item \textbf{Oui} à "Register Anaconda3 as my default Python"
			\end{enumerate}
			
			\item Cliquez sur \textbf{Install}
			\begin{itemize}
				\item L'installation commence. Ça prend 5-10 minutes.
				\item C'est le moment de faire une prière aux dieux du code.
			\end{itemize}
			
			\item À la fin, cliquez sur \textbf{Next}, puis \textbf{Finish}.
		\end{enumerate}
	\end{installation}

	\section{Les pièges du débutant}
	
	\section{Piège n°1 : "Anaconda Prompt" vs "Terminal normal"}
	
	\textbf{Symptôme :} Vous ouvrez le bon vieux terminal noir (cmd.exe) et vous tapez \texttt{conda --version} $\rightarrow$ \textit{BOUM} erreur.
	
	\textbf{Raison :} Le terminal normal ne connaît pas Anaconda. C'est comme demander à un poisson de monter à un arbre.
	
	\textbf{Solution :} Utilisez \textbf{toujours} le "Anaconda Prompt" sur Windows. C'est comme un terminal normal, mais il a mangé des légumes.
	
	\begin{figure}[H]
		\centering
		%\includegraphics[width=0.7\textwidth]{erreur_conda.png}
		\caption{Erreur "conda n'est pas reconnu" dans le mauvais terminal}

	\end{figure}
	
	\begin{figure}[H]

		%\includegraphics[width=0.7\textwidth]{bon_termminal.png}
		\caption{Utilisez le bon terminal - Anaconda Prompt}
		\label{fig:bon_terminal}
		\textbf{Légende :} Le Anaconda Prompt fonctionne correctement.
	\end{figure}
	
	\section{Piège n°2 : "J'ai coché PATH et maintenant tout est cassé"}
	
	\textbf{Symptôme :} Vous avez un ancien Python installé. Vous avez installé Anaconda. Quand vous tapez \texttt{python}, il lance la mauvaise version. Vous avez l'impression d'avoir deux méchants qui se battent dans votre ordinateur.
	
	\textbf{Solution :}
	\begin{enumerate}
		\item Allez dans : \texttt{Panneau de configuration > Systeme > Parametres systeme avances}
		\item Cliquez sur "Variables d'environnement"
		\item Dans la section "Variables système", cherchez "Path"
		\item Double-cliquez dessus
		\item Assurez-vous que le chemin vers Anaconda (\texttt{C:\textbackslash Users\textbackslash VotreNom\textbackslash anaconda3}) est \textbf{EN HAUT} de la liste
		\item Supprimez les autres chemins Python si il y en a
	\end{enumerate}
	
	\begin{figure}[H]
		\centering
		%\includegraphics[width=0.7\textwidth]{path_variables.png}
		\caption{Modification des variables d'environnement PATH}
		\label{fig:path_vars}
		\textbf{Légende :} Assurez-vous que le chemin Anaconda est en haut de la liste.
	\end{figure}
	
	\section{Piège n°3 : "J'ai installé Anaconda, mais j'ai plus de place sur mon disque dur"}
	
	\textbf{Symptôme :} Vous avez installé Anaconda (c'est gros, \textasciitilde 600 Mo), puis vous avez installé TensorFlow (encore gros), puis PyTorch (encore plus gros). Votre disque dur pleure.
	
	\textbf{Raison :} Anaconda, c'est un frigo géant. Ça prend de la place.
	
	\textbf{Solution :} 
	\begin{itemize}
		\item N'installez pas toutes les bibliothèques en même temps
		\item Utilisez des environnements (on en parle juste après)
		\item Acceptez que votre disque dur va souffrir. C'est le prix du progrès.
	\end{itemize}
	
	\section{Piège n°4 : "J'ai tout installé mais j'arrive pas à importer pandas"}
	
	\textbf{Symptôme :} 
	\begin{lstlisting}[language=Python]
		import pandas
		# ModuleNotFoundError: No module named 'pandas'
	\end{lstlisting}
	
	\textbf{Raison :} Pandas n'est pas installé (sauf que si, normalement Anaconda l'a installé, mais vous avez peut-être un environnement différent).
	
	\textbf{Solution :}
	\begin{lstlisting}[language=bash]
		conda install pandas
	\end{lstlisting}
	
	C'est comme dire au vendeur : "He, donne-moi des pandas !"

	\section{Les environnements conda}
	
	Imaginez que vous êtes à la cantine. Vous avez un plateau. Vous mettez du riz et du poisson dessus. Tout va bien.
	
	Maintenant, imaginez que vous avez 10 amis qui viennent manger. Chacun a ses propres goûts. Si vous mettez tout sur le même plateau, c'est le bazar.
	
	\textbf{Les environnements conda, c'est comme donner un plateau séparé à chaque ami.}
	
	\begin{definition}
		Un \textbf{environnement conda} est un espace isolé où vous pouvez installer différentes versions de Python et de bibliothèques sans affecter les autres environnements.
	\end{definition}
	
	\section*{Commandes essentielles}
	
	\begin{installation}
		\textbf{Créer un environnement}
		\begin{lstlisting}[language=bash]
			conda create --name projet_ia python=3.11
		\end{lstlisting}
		C'est comme dire : "Je crée une nouvelle cuisine avec Python 3.11 et rien d'autre dedans."
		
		\textbf{Activer un environnement}
		\begin{lstlisting}[language=bash]
			conda activate projet_ia
		\end{lstlisting}
		C'est comme dire : "Je vais travailler dans cette cuisine maintenant."
		
		\textbf{Installer une bibliothèque dans un environnement}
		\begin{lstlisting}[language=bash]
			conda activate projet_ia
			conda install numpy pandas matplotlib
		\end{lstlisting}
		
		\textbf{Voir tous les environnements}
		\begin{lstlisting}[language=bash]
			conda env list
		\end{lstlisting}
		
		\textbf{Désactiver un environnement}
		\begin{lstlisting}[language=bash]
			conda deactivate
		\end{lstlisting}
		
		\textbf{Supprimer un environnement}
		\begin{lstlisting}[language=bash]
			conda env remove --name projet_ia
		\end{lstlisting}
	\end{installation}

	\section{Troubleshooting}
	
	\section{Erreur : "conda n'est pas reconnu"}
	
	\textbf{Solution :}
	\begin{itemize}
		\item Sur Windows : Ouvrez le "Anaconda Prompt" (pas le terminal normal)
		\item Sur Mac/Linux : Tapez \texttt{source \textasciitilde/.bashrc} ou \texttt{source \textasciitilde/.zshrc}
	\end{itemize}
	
	\section{Erreur : "Python n'est pas reconnu"}
	
	\textbf{Solution :}
	\begin{itemize}
		\item Réinstallez Anaconda (c'est radical mais ça marche)
		\item Vérifiez le PATH (voir section plus haut)
	\end{itemize}
	
	\section{Erreur : "Permission denied"}
	
	\textbf{Solution :}
	\begin{itemize}
		\item Sur Mac/Linux : Utilisez \texttt{sudo} devant la commande
		\item Sur Windows : Lancez l'installateur en tant qu'administrateur
	\end{itemize}
	
	\section{Mon ordinateur est lent !}
	
	\textbf{Solution :}
	\begin{itemize}
		\item Redémarrez l'ordinateur (oui, vraiment)
		\item Ne lancez pas Chrome, VS Code, Spotify, et Jupyter en même temps
		\item Acceptez que la programmation, c'est comme une course de Formule 1 : ça consomme de l'énergie
	\end{itemize}
